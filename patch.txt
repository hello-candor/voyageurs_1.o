                                                            </p>
                                                            <div className="flex items-center justify-end gap-2">
                                                                <DollarSign size={16} className={iPaid ? 'text-med-olive' : iOwe ? 'text-med-terracotta' : 'text-gray-400'} />
                                                                <span className={`font-serif text-3xl ${iPaid ? 'text-med-olive' : iOwe ? 'text-med-terracotta' : 'text-gray-600 dark:text-gray-300'}`}>
                                                                    {(iPaid ? (sharePerPerson * exp.splitWithIds.length) : iOwe ? sharePerPerson : exp.amount).toFixed(2)}
                                                                </span>
                                                            </div>
                                                        </div>
                                                    </div>

                                                    <div className="mt-6 pt-6 border-t border-gray-100 dark:border-gray-800 relative z-10 flex flex-col md:flex-row md:items-center justify-between gap-4">
                                                        <div className="flex items-center gap-4 text-[10px] font-bold uppercase text-gray-400">
                                                            <div className="flex -space-x-2">
                                                                {/* Mock avatars for splitters */}
                                                                <div className="w-8 h-8 rounded-full bg-gray-200 border-2 border-white dark:border-gray-900 flex items-center justify-center text-[10px] text-gray-500 shadow-sm z-30">
                                                                    {exp.payerName.charAt(0)}
                                                                </div>
                                                                {exp.splitWithIds.slice(0, 3).map((id, i) => (
                                                                    <div key={i} className={`w-8 h-8 rounded-full bg-gray-200 border-2 border-white dark:border-gray-900 flex items-center justify-center text-[10px] text-gray-500 shadow-sm z-${20-i}`}>
                                                                        {allGuests.find(g => g.email === id)?.name?.charAt(0) || '?'}
                                                                    </div>
                                                                ))}
                                                                {exp.splitWithIds.length > 3 && (
                                                                    <div className="w-8 h-8 rounded-full bg-gray-100 border-2 border-white dark:border-gray-900 flex items-center justify-center text-[10px] text-gray-500 shadow-sm z-0">
                                                                        +{exp.splitWithIds.length - 3}
                                                                    </div>
                                                                )}
                                                            </div>
                                                            <span className="tracking-widest">{exp.splitWithIds.length + 1} People</span>
                                                        </div>

                                                        {/* Action Buttons */}
                                                        {viewStatus === 'active' && iOwe && (
                                                            <div className="flex gap-2">
                                                                {getPaymentLink(allGuests.find(g => g.email === exp.payerId)!, sharePerPerson) ? (
                                                                    <a 
                                                                        href={getPaymentLink(allGuests.find(g => g.email === exp.payerId)!, sharePerPerson)!}
                                                                        target="_blank" rel="noopener noreferrer"
                                                                        className="flex-1 px-4 py-2 bg-med-blue text-white rounded-xl text-[10px] font-bold uppercase tracking-widest hover:bg-med-blue/90 transition-colors flex items-center justify-center gap-2 shadow-sm"
                                                                    >
                                                                        <Smartphone size={14} /> Pay Now
                                                                    </a>
                                                                ) : (
                                                                    <span className="flex-1 px-4 py-2 bg-gray-100 dark:bg-gray-800 text-gray-400 rounded-xl text-[10px] font-bold uppercase tracking-widest text-center">
                                                                        No Pay Link
                                                                    </span>
                                                                )}
                                                                
                                                                <button 
                                                                    onClick={() => resolveSharedExpense(exp.id)}
                                                                    className="px-4 py-2 border border-gray-200 dark:border-gray-700 text-gray-600 dark:text-gray-300 rounded-xl text-[10px] font-bold uppercase tracking-widest hover:bg-gray-50 dark:hover:bg-gray-800 transition-colors shadow-sm"
                                                                >
                                                                    Mark Paid
                                                                </button>
                                                            </div>
                                                        )}
                                                        {viewStatus === 'active' && iPaid && (
                                                             <button 
                                                                onClick={() => resolveSharedExpense(exp.id)}
                                                                className="px-6 py-2 bg-med-olive text-white rounded-xl text-[10px] font-bold uppercase tracking-widest hover:bg-med-olive/90 transition-colors flex items-center justify-center gap-2 shadow-sm"
                                                            >
                                                                <CheckCircle2 size={14} /> Mark Settled
                                                            </button>
                                                        )}
                                                    </div>
                                                    
                                                    {/* Decorative Background Icon */}
                                                    <div className="absolute -right-6 -bottom-6 opacity-5 rotate-12 pointer-events-none">
                                                        <catInfo.icon size={160} />
                                                    </div>
                                                </div>
                                            );
                                        })
                                    )}
                                </div>
                            </div>
                        )}
                    </div>
                </div>
            </div>
        </div>
    );
};
